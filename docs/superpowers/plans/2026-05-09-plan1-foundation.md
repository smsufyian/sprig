# sprig — Plan 1: Foundation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A working `sprig` CLI where `sprig init`, `sprig create <name>`, `sprig list`, `sprig status <name>`, `sprig start <name>`, `sprig stop <name>`, `sprig destroy <name>`, `sprig open <name>`, `sprig config edit`, `sprig version`, `sprig prune`, `sprig completion`, and `sprig telemetry` all function end-to-end. Space creation is real (state persisted in `~/.sprig/registry.json`, ports allocated), but no actual containers are created yet — that is Plan 2.

**Architecture:** Cobra CLI commands call an `Engine` (all methods accept `context.Context`) which orchestrates a `Registry` (JSON at `~/.sprig/registry.json`), a `PortAllocator` (JSON at `~/.sprig/ports.json`), and a `platform.Driver` interface. Every registry/ports write is protected by `lock.WithLock`. In this plan the driver is a no-op stub. `config.Load` is a hand-rolled TOML parser (no reflection). `config.Merge` applies overrides on top of a detected baseline (stub in this plan). All commands support `--output json`. Signal handling cancels in-flight operations and cleans up partial state.

**Tech Stack:** Go 1.23, `github.com/spf13/cobra v1.8.1`, `github.com/mailru/easyjson v0.7`, `github.com/posthog/posthog-go v1`

```go
// internal/config/config.go
package config

import (
	"os"

	"github.com/BurntSushi/toml"
)

// Config is the merged result of auto-detected stack + sprig.toml overrides.
type Config struct {
	Project  ProjectConfig            `toml:"project"`
	Services map[string]ServiceConfig `toml:"services"`
	DB       DBConfig                 `toml:"db"`
}

// ProjectConfig holds top-level project settings.
type ProjectConfig struct {
	Name string `toml:"name"`
	Run  string `toml:"run"`
}

// ServiceConfig describes a single service (postgres, redis, kafka, etc.).
type ServiceConfig struct {
	Version    string            `toml:"version"`
	Enabled    *bool             `toml:"enabled"`
	Port       int               `toml:"port"`
	Env        map[string]string `toml:"env"`
	Partitions int               `toml:"partitions"` // kafka
	Memory     string            `toml:"memory"`     // elasticsearch
}

// DBConfig holds database-level settings.
type DBConfig struct {
	Migrations string `toml:"migrations"`
}

// Load reads sprig.toml from path. A missing file is not an error — it returns
// an empty Config so detection-only workflows work without a toml file.
func Load(path string) (Config, error) {
	var cfg Config
	if _, err := os.Stat(path); os.IsNotExist(err) {
		return cfg, nil
	}
	_, err := toml.DecodeFile(path, &cfg)
	return cfg, err
}

// Merge applies override on top of base, field by field. Zero/nil values in
// override do not overwrite non-zero values in base.
func Merge(base, override Config) Config {
	if override.Project.Name != "" {
		base.Project.Name = override.Project.Name
	}
	if override.Project.Run != "" {
		base.Project.Run = override.Project.Run
	}
	if override.DB.Migrations != "" {
		base.DB.Migrations = override.DB.Migrations
	}
	if base.Services == nil {
		base.Services = make(map[string]ServiceConfig)
	}
	for name, ov := range override.Services {
		ex := base.Services[name]
		if ov.Version != "" {
			ex.Version = ov.Version
		}
		if ov.Enabled != nil {
			ex.Enabled = ov.Enabled
		}
		if ov.Port != 0 {
			ex.Port = ov.Port
		}
		if ov.Partitions != 0 {
			ex.Partitions = ov.Partitions
		}
		if ov.Memory != "" {
			ex.Memory = ov.Memory
		}
		if len(ov.Env) > 0 {
			if ex.Env == nil {
				ex.Env = make(map[string]string)
			}
			for k, v := range ov.Env {
				ex.Env[k] = v
			}
		}
		base.Services[name] = ex
	}
	return base
}
```

```go
// internal/registry/registry.go
package registry

import (
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"time"
)

// Status is the lifecycle state of a space.
type Status string

const (
	StatusCreating Status = "creating"
	StatusRunning  Status = "running"
	StatusStopped  Status = "stopped"
	StatusDegraded Status = "degraded"
)

// Space is a single isolated virtual environment.
type Space struct {
	Name    string         `json:"name"`
	Project string         `json:"project"`
	Status  Status         `json:"status"`
	Created time.Time      `json:"created"`
	Ports   map[string]int `json:"ports"`
	Stack   StackSummary   `json:"stack"`
}

// StackSummary is a human-readable summary of the detected stack.
type StackSummary struct {
	Runtime  string   `json:"runtime"`
	Services []string `json:"services"`
}

type store struct {
	Spaces map[string]Space `json:"spaces"`
}

// Registry persists spaces to a JSON file.
type Registry struct {
	path string
}

// New creates a Registry backed by dir/registry.json. Creates dir if needed.
func New(dir string) (*Registry, error) {
	if err := os.MkdirAll(dir, 0755); err != nil {
		return nil, fmt.Errorf("create sprig dir: %w", err)
	}
	return &Registry{path: filepath.Join(dir, "registry.json")}, nil
}

func (r *Registry) load() (store, error) {
	s := store{Spaces: make(map[string]Space)}
	b, err := os.ReadFile(r.path)
	if os.IsNotExist(err) {
		return s, nil
	}
	if err != nil {
		return s, err
	}
	return s, json.Unmarshal(b, &s)
}

func (r *Registry) save(s store) error {
	b, err := json.MarshalIndent(s, "", "  ")
	if err != nil {
		return err
	}
	tmp := r.path + ".tmp"
	if err := os.WriteFile(tmp, b, 0644); err != nil {
		return err
	}
	return os.Rename(tmp, r.path)
}

// Add inserts a new space. Returns an error if a space with the same name exists.
func (r *Registry) Add(sp Space) error {
	s, err := r.load()
	if err != nil {
		return err
	}
	if _, exists := s.Spaces[sp.Name]; exists {
		return fmt.Errorf("space %q already exists", sp.Name)
	}
	s.Spaces[sp.Name] = sp
	return r.save(s)
}

// Get returns the space with the given name.
func (r *Registry) Get(name string) (Space, error) {
	s, err := r.load()
	if err != nil {
		return Space{}, err
	}
	sp, ok := s.Spaces[name]
	if !ok {
		return Space{}, fmt.Errorf("space %q not found", name)
	}
	return sp, nil
}

// Update applies fn to the named space and persists the result.
func (r *Registry) Update(name string, fn func(*Space)) error {
	s, err := r.load()
	if err != nil {
		return err
	}
	sp, ok := s.Spaces[name]
	if !ok {
		return fmt.Errorf("space %q not found", name)
	}
	fn(&sp)
	s.Spaces[name] = sp
	return r.save(s)
}

// Remove deletes the named space.
func (r *Registry) Remove(name string) error {
	s, err := r.load()
	if err != nil {
		return err
	}
	if _, ok := s.Spaces[name]; !ok {
		return fmt.Errorf("space %q not found", name)
	}
	delete(s.Spaces, name)
	return r.save(s)
}

// List returns all registered spaces in unspecified order.
func (r *Registry) List() ([]Space, error) {
	s, err := r.load()
	if err != nil {
		return nil, err
	}
	spaces := make([]Space, 0, len(s.Spaces))
	for _, sp := range s.Spaces {
		spaces = append(spaces, sp)
	}
	return spaces, nil
}
```

```go
// internal/registry/ports.go
package registry

import (
	"encoding/json"
	"os"
	"path/filepath"
)

const basePort = 52000

type portStore struct {
	Allocations map[string]map[string]int `json:"allocations"`
	NextPort    int                       `json:"next_port"`
}

// PortAllocator assigns non-overlapping host ports to space services.
type PortAllocator struct {
	path string
}

// NewPortAllocator creates a PortAllocator backed by dir/ports.json.
func NewPortAllocator(dir string) (*PortAllocator, error) {
	if err := os.MkdirAll(dir, 0755); err != nil {
		return nil, err
	}
	return &PortAllocator{path: filepath.Join(dir, "ports.json")}, nil
}

func (p *PortAllocator) load() (portStore, error) {
	d := portStore{
		Allocations: make(map[string]map[string]int),
		NextPort:    basePort,
	}
	b, err := os.ReadFile(p.path)
	if os.IsNotExist(err) {
		return d, nil
	}
	if err != nil {
		return d, err
	}
	return d, json.Unmarshal(b, &d)
}

func (p *PortAllocator) save(d portStore) error {
	b, err := json.MarshalIndent(d, "", "  ")
	if err != nil {
		return err
	}
	tmp := p.path + ".tmp"
	if err := os.WriteFile(tmp, b, 0644); err != nil {
		return err
	}
	return os.Rename(tmp, p.path)
}

// Allocate returns a map of service → host port for the given space.
// If the space already has an allocation it is returned unchanged (idempotent).
func (p *PortAllocator) Allocate(spaceName string, services []string) (map[string]int, error) {
	d, err := p.load()
	if err != nil {
		return nil, err
	}
	if existing, ok := d.Allocations[spaceName]; ok {
		return existing, nil
	}
	ports := make(map[string]int, len(services))
	for _, svc := range services {
		ports[svc] = d.NextPort
		d.NextPort++
	}
	d.Allocations[spaceName] = ports
	return ports, p.save(d)
}

// Release removes all port allocations for the given space.
func (p *PortAllocator) Release(spaceName string) error {
	d, err := p.load()
	if err != nil {
		return err
	}
	delete(d.Allocations, spaceName)
	return p.save(d)
}
```

```go
// internal/engine/engine.go
package engine

import (
	"fmt"
	"time"

	"github.com/smsufyian/sprig/internal/config"
	"github.com/smsufyian/sprig/internal/platform"
	"github.com/smsufyian/sprig/internal/registry"
)

// SpaceOptions holds parameters for creating a space.
type SpaceOptions struct {
	Name    string
	Project string // absolute path to the project directory
	From    string // seed source: "production", a named connection, or "" for empty
	CIMode  bool
}

// Engine orchestrates the space lifecycle.
type Engine struct {
	registry *registry.Registry
	ports    *registry.PortAllocator
	driver   platform.Driver
	sprigDir string
}

// New creates an Engine from its dependencies.
func New(reg *registry.Registry, ports *registry.PortAllocator, driver platform.Driver, sprigDir string) *Engine {
	return &Engine{
		registry: reg,
		ports:    ports,
		driver:   driver,
		sprigDir: sprigDir,
	}
}

// enabledServices returns the list of service names to provision from cfg.
// Always includes "app". Plan 3 replaces this with real stack detection.
func enabledServices(cfg config.Config) []string {
	services := []string{"app"}
	for name, svc := range cfg.Services {
		if svc.Enabled == nil || *svc.Enabled {
			services = append(services, name)
		}
	}
	if len(cfg.Services) == 0 {
		// Default stub services when no config is present.
		services = append(services, "postgres")
	}
	return services
}

// Create provisions a new space and registers it.
func (e *Engine) Create(opts SpaceOptions, cfg config.Config) (*registry.Space, error) {
	services := enabledServices(cfg)

	ports, err := e.ports.Allocate(opts.Name, services)
	if err != nil {
		return nil, fmt.Errorf("allocate ports: %w", err)
	}

	sp := registry.Space{
		Name:    opts.Name,
		Project: opts.Project,
		Status:  registry.StatusRunning,
		Created: time.Now(),
		Ports:   ports,
		Stack: registry.StackSummary{
			Runtime:  cfg.Project.Name,
			Services: services,
		},
	}

	if err := e.registry.Add(sp); err != nil {
		_ = e.ports.Release(opts.Name)
		return nil, fmt.Errorf("register space: %w", err)
	}

	return &sp, nil
}

// Destroy removes a space and releases its port allocation.
func (e *Engine) Destroy(name string) error {
	if _, err := e.registry.Get(name); err != nil {
		return err
	}
	if err := e.registry.Remove(name); err != nil {
		return err
	}
	return e.ports.Release(name)
}

// Status returns the current state of a space.
func (e *Engine) Status(name string) (registry.Space, error) {
	return e.registry.Get(name)
}

// List returns all spaces across all projects.
func (e *Engine) List() ([]registry.Space, error) {
	return e.registry.List()
}

// Start marks a stopped space as running.
func (e *Engine) Start(name string) error {
	return e.registry.Update(name, func(s *registry.Space) {
		s.Status = registry.StatusRunning
	})
}

// Stop marks a running space as stopped.
func (e *Engine) Stop(name string) error {
	return e.registry.Update(name, func(s *registry.Space) {
		s.Status = registry.StatusStopped
	})
}
```

```go
// printPorts prints service URLs in a human-readable format.
func printPorts(ports map[string]int, spaceName string) {
	for svc, port := range ports {
		switch svc {
		case "app":
			fmt.Printf("  %-12s  http://localhost:%d\n", "API", port)
		case "postgres":
			fmt.Printf("  %-12s  postgresql://localhost:%d/%s\n", "DB", port, spaceName)
		default:
			fmt.Printf("  %-12s  localhost:%d\n", svc, port)
		}
	}
}

// toEnvKey converts a service name to an uppercase env var key segment.
func toEnvKey(s string) string {
	out := make([]byte, len(s))
	for i := range s {
		if s[i] >= 'a' && s[i] <= 'z' {
			out[i] = s[i] - 32
		} else {
			out[i] = s[i]
		}
	}
	return string(out)
}
```

```go
// internal/lock/lock.go
package lock

import (
	"fmt"
	"os"
	"syscall"
	"time"
)

const lockTimeout = 5 * time.Second
const lockPollInterval = 50 * time.Millisecond

// WithLock acquires an exclusive flock on path+".lock", calls fn, then releases.
// Returns an error if the lock cannot be acquired within 5 seconds.
func WithLock(path string, fn func() error) error {
	lockPath := path + ".lock"
	f, err := os.OpenFile(lockPath, os.O_CREATE|os.O_RDWR, 0600)
	if err != nil {
		return fmt.Errorf("open lock file: %w", err)
	}
	defer f.Close()

	deadline := time.Now().Add(lockTimeout)
	for {
		err := syscall.Flock(int(f.Fd()), syscall.LOCK_EX|syscall.LOCK_NB)
		if err == nil {
			break
		}
		if err != syscall.EWOULDBLOCK {
			return fmt.Errorf("acquire lock: %w", err)
		}
		if time.Now().After(deadline) {
			return fmt.Errorf("another sprig process is running — retry in a moment")
		}
		time.Sleep(lockPollInterval)
	}
	defer syscall.Flock(int(f.Fd()), syscall.LOCK_UN) //nolint:errcheck

	return fn()
}
```

```go
// internal/validate/space.go
package validate

import (
	"fmt"
	"strings"
)

var reserved = map[string]bool{
	"base":      true,
	"default":   true,
	"snapshots": true,
}

// SpaceName returns an error if name is not a valid sprig space name.
// Valid: starts with [a-z], contains only [a-z0-9-], max 40 chars.
func SpaceName(name string) error {
	if name == "" {
		return fmt.Errorf("space name must not be empty")
	}
	if len(name) > 40 {
		return fmt.Errorf("space name %q exceeds 40 characters", name)
	}
	if reserved[name] {
		return fmt.Errorf("space name %q is reserved", name)
	}
	if name[0] < 'a' || name[0] > 'z' {
		return fmt.Errorf("space name %q must start with a lowercase letter", name)
	}
	for _, c := range name {
		if !((c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') || c == '-') {
			return fmt.Errorf("space name %q must contain only [a-z0-9-], got %q", name, strings.ToUpper(string(c)))
		}
	}
	return nil
}
```

```go
package telemetry

import (
	"encoding/json"
	"os"
	"path/filepath"
	"time"

	"github.com/posthog/posthog-go"
)

const posthogAPIKey = "phc_sprig_placeholder" // replace with real key at release

type state struct {
	Disabled bool   `json:"disabled"`
	NodeID   string `json:"node_id"`
}

// Client wraps PostHog with opt-out and anonymous ID management.
type Client struct {
	ph     posthog.Client
	nodeID string
	active bool
}

// New loads telemetry state from sprigDir/telemetry.json and initialises the client.
func New(sprigDir string) (*Client, error) {
	if os.Getenv("SPRIG_NO_TELEMETRY") != "" {
		return &Client{active: false}, nil
	}

	statePath := filepath.Join(sprigDir, "telemetry.json")
	st, err := loadState(statePath)
	if err != nil {
		return &Client{active: false}, nil
	}
	if st.Disabled {
		return &Client{active: false}, nil
	}

	ph, err := posthog.NewWithConfig(posthogAPIKey, posthog.Config{
		Endpoint: "https://app.posthog.com",
	})
	if err != nil {
		return &Client{active: false}, nil
	}

	return &Client{ph: ph, nodeID: st.NodeID, active: true}, nil
}

// Track fires an event asynchronously. Never blocks.
func (c *Client) Track(event string, props map[string]any) {
	if !c.active {
		return
	}
	props["$lib"] = "sprig"
	go c.ph.Enqueue(posthog.Capture{ //nolint:errcheck
		DistinctId: c.nodeID,
		Event:      event,
		Properties: posthog.NewProperties().SetAll(props),
		Timestamp:  time.Now(),
	})
}

// Close flushes pending events. Call on process exit.
func (c *Client) Close() {
	if c.active && c.ph != nil {
		c.ph.Close()
	}
}

// Disable writes opt-out to telemetry.json.
func Disable(sprigDir string) error {
	return writeState(filepath.Join(sprigDir, "telemetry.json"), func(st *state) { st.Disabled = true })
}

// Enable removes opt-out from telemetry.json.
func Enable(sprigDir string) error {
	return writeState(filepath.Join(sprigDir, "telemetry.json"), func(st *state) { st.Disabled = false })
}

func loadState(path string) (state, error) {
	var st state
	b, err := os.ReadFile(path)
	if os.IsNotExist(err) {
		st.NodeID = generateID()
		return st, writeRawState(path, st)
	}
	if err != nil {
		return st, err
	}
	return st, json.Unmarshal(b, &st)
}

func writeState(path string, fn func(*state)) error {
	st, _ := loadState(path)
	fn(&st)
	return writeRawState(path, st)
}

func writeRawState(path string, st state) error {
	b, err := json.MarshalIndent(st, "", "  ")
	if err != nil {
		return err
	}
	return os.WriteFile(path, b, 0600)
}

func generateID() string {
	b := make([]byte, 16)
	// Use time-based seed for a simple unique ID — no crypto needed here.
	t := time.Now().UnixNano()
	for i := range b {
		b[i] = byte(t>>(uint(i)*4)) ^ byte(i*31)
	}
	return fmt.Sprintf("%x", b)
}
```

```go
// internal/cli/prune.go
package cli

import (
	"bufio"
	"fmt"
	"os"
	"strings"
	"time"

	"github.com/spf13/cobra"
	"github.com/smsufyian/sprig/internal/registry"
)

func newPruneCmd() *cobra.Command {
	var days int
	var dryRun bool

	cmd := &cobra.Command{
		Use:   "prune",
		Short: "Remove stopped or stale spaces",
		RunE: func(cmd *cobra.Command, args []string) error {
			ctx := cmd.Context()
			eng, err := newEngine()
			if err != nil {
				return err
			}
			spaces, err := eng.List(ctx)
			if err != nil {
				return err
			}

			cutoff := time.Now().AddDate(0, 0, -days)
			var candidates []registry.Space
			for _, s := range spaces {
				if s.Status == registry.StatusStopped && s.Created.Before(cutoff) {
					candidates = append(candidates, s)
				}
			}

			if len(candidates) == 0 {
				fmt.Println("No stale spaces found.")
				return nil
			}

			fmt.Printf("Stale spaces (stopped for >%d days):\n\n", days)
			for _, s := range candidates {
				fmt.Printf("  %-20s  stopped  project: %s\n", s.Name, s.Project)
			}
			fmt.Println()

			if dryRun {
				fmt.Printf("Dry run — %d space(s) would be destroyed.\n", len(candidates))
				return nil
			}

			fmt.Printf("Destroy %d space(s)? [y/N] ", len(candidates))
			reader := bufio.NewReader(os.Stdin)
			answer, _ := reader.ReadString('\n')
			if strings.ToLower(strings.TrimSpace(answer)) != "y" {
				fmt.Println("Aborted.")
				return nil
			}

			for _, s := range candidates {
				if err := eng.Destroy(ctx, s.Name); err != nil {
					fmt.Fprintf(os.Stderr, "error destroying %s: %v\n", s.Name, err)
					continue
				}
				fmt.Printf("  destroyed %s\n", s.Name)
			}
			return nil
		},
	}

	cmd.Flags().IntVar(&days, "days", 3, "remove spaces stopped for more than N days")
	cmd.Flags().BoolVar(&dryRun, "dry-run", false, "show what would be removed without acting")
	return cmd
}
```
